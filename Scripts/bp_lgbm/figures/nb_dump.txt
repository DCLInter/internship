LW          = 1.8

def get_r2(df, target, subset, x_vals):
    """Return R² (%) list aligned to x_vals; NaN if an n_subjects value is missing."""
    sub = df[(df["target"] == target) & (df["subset"] == subset)].set_index("n_subjects")
    return [sub.loc[x, "R2"] * 100 if x in sub.index else np.nan for x in x_vals]


def _add_labels(ax, xi, vals, va, y_off, x_off, fs):
    for i, v in enumerate(vals):
        if not np.isnan(v):
            ax.text(
                xi[i] + x_off, v + y_off,
                f"{v:.2f}%",
                ha="center", va=va,
                fontsize=fs, fontfamily=FONT_FAMILY,
                color=COLOR_LABEL,
            )


def make_subject_size_fig(width_in: float, target: str = TARGET):
    height_in = width_in * ASPECT
    is_small  = width_in < 5
    label_fs  = 5.5 if is_small else 7.0
    tick_fs   = 6   if is_small else 8
    title_fs  = 7   if is_small else 10
    leg_fs    = 6   if is_small else 8
    lw        = 1.2 if is_small else LW
    ms        = 3.5 if is_small else MS

    xi = np.arange(len(N_SUBJECTS))

    fig, ax = plt.subplots(
        figsize=(width_in, height_in),
        dpi=DPI,
        layout="constrained",
    )

    gen_tr = get_r2(df_gen, target, "Train", N_SUBJECTS)
    gen_te = get_r2(df_gen, target, "Test",  N_SUBJECTS)
    fb_tr  = get_r2(df_fb,  target, "Train", N_SUBJECTS)
    fb_te  = get_r2(df_fb,  target, "Test",  N_SUBJECTS)

    # Reference line at y=0
    ax.axhline(0, color="#888888", lw=0.9, ls="-", zorder=2)

    # Lines: filled markers = Training, hollow markers = Testing
    ax.plot(xi, gen_tr, color=COLOR_GEN, ls="-",  marker="o", lw=lw, ms=ms, zorder=3)
